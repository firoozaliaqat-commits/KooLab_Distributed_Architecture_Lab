"""API and browser-asset smoke tests. Run: pytest -q"""
import hashlib, importlib, json, os, sys, tempfile
from pathlib import Path
from fastapi.testclient import TestClient

DATA=tempfile.mkdtemp(prefix='koolab-v2-')
os.environ['LAB_DATA']=DATA
os.environ['APP_SIGNING_SECRET']='test-only-signing-key'*4
os.environ['APP_ADMIN_HASH']=hashlib.pbkdf2_hmac('sha256',b'test-password',b'koolab-admin-v2',230000).hex()
from app.main import app
client=TestClient(app)

def login():
    res=client.post('/api/auth/login',json={'password':'test-password'})
    assert res.status_code==200, res.text
    return {'Authorization':'Bearer '+res.json()['token']}

def test_dashboard_boot_and_headers():
    r=client.get('/')
    assert r.status_code==200
    assert b'KooLab CloudLab Pro' in r.content
    assert 'nosniff' == r.headers['x-content-type-options']
    assert 'frame-ancestors' in r.headers['content-security-policy']
    for p in ['/static/app.js','/static/style.css','/healthz','/api/status','/api/identity']:
        assert client.get(p).status_code==200,p

def test_auth_required_and_password_validation():
    assert client.post('/api/messages',json={'body':'x'}).status_code==401
    assert client.post('/api/stress/start',json={'seconds':15,'workers':1}).status_code==401
    assert client.post('/api/probe/upload',content=b'x').status_code==401
    assert client.post('/api/auth/login',json={'password':'incorrect'}).status_code==401
    h=login()
    assert client.get('/api/auth/me',headers=h).status_code==200
    assert client.get('/api/auth/me',headers={'Authorization':'Bearer garbage'}).status_code==401

def test_sqlite_messages_crud_and_injection_string_is_data():
    h=login()
    test_text="SQL injection test: '; DROP TABLE lab_messages; --"
    r=client.post('/api/messages',headers=h,json={'body':test_text})
    assert r.status_code==200,r.text
    r=client.get('/api/messages')
    assert r.status_code==200,r.text
    assert any(x['body']==test_text for x in r.json()['items'])
    assert client.post('/api/messages',headers=h,json={'body':'x'*251}).status_code==422

def test_rds_wizard_and_security():
    h=login()
    assert client.get('/api/rds/config').status_code==401
    c=client.get('/api/rds/config',headers=h).json()
    assert c['configured'] is False
    assert client.get('/api/rds/verify',headers=h).json()['mode']=='sqlite'
    assert client.post('/api/rds/test',headers=h,json={'host':'8.8.8.8','password':'dummy'}).status_code==422
    r=client.get('/api/security/checks',headers=h)
    assert r.status_code==200
    assert any(x['id']=='sg' and x['result']=='manual' for x in r.json()['checks'])
    r=client.get('/api/self-test',headers=h).json()
    assert any(x['name']=='Auto Scaling triggered' and x['passed'] is None for x in r['tests'])

def test_bandwidth_bounds():
    h=login()
    assert client.get('/api/probe/download?kb=999',headers=h).status_code==422
    r=client.get('/api/probe/download?kb=1',headers=h)
    assert r.status_code==200 and len(r.content)==1024
    r=client.post('/api/probe/upload',headers=h,content=b'X'*1024)
    assert r.status_code==200 and r.json()['received_bytes']==1024
    assert client.post('/api/probe/upload',headers=h,content=b'X'*(512*1024+1)).status_code==413

def test_stress_validation_and_lifecycle():
    h=login()
    assert client.post('/api/stress/start',headers=h,json={'seconds':1000,'workers':3}).status_code==422
    r=client.post('/api/stress/start',headers=h,json={'seconds':15,'workers':1})
    assert r.status_code==200,r.text
    assert r.json()['running'] is True
    assert client.post('/api/stress/start',headers=h,json={'seconds':15,'workers':1}).status_code in (409,429)
    assert client.post('/api/stress/stop',headers=h).status_code==200
