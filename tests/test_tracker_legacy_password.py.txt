"""KooLab HTTP classroom reporter tests. No requests to actual instructor tracker."""
import os, tempfile, hashlib
from fastapi.testclient import TestClient
os.environ.setdefault('LAB_DATA',tempfile.mkdtemp(prefix='koolab-v5-tests-'))
os.environ.setdefault('APP_SIGNING_SECRET','test-only-signing-key'*4)
os.environ.setdefault('APP_ADMIN_HASH',hashlib.pbkdf2_hmac('sha256',b'test-password',b'koolab-admin-v2',230000).hex())
from app import main, tracker
client=TestClient(main.app)
TOKEN='temporary-classroom-student-token-'+('X'*32)
ORIGIN='http://testserver'

def login():
    main.RATES.clear()
    r=client.post('/api/auth/login',json={'password':'test-password'})
    assert r.status_code==200,r.text
    return {'Authorization':'Bearer '+r.json()['token'],'Origin':ORIGIN}

def test_health():
    assert client.get('/healthz').status_code==200

def test_tracker_requires_lab_auth():
    assert client.post('/api/tracker/me',headers={'Origin':ORIGIN},json={'student_token':TOKEN}).status_code==401

def test_http_student_link_and_start(monkeypatch):
    calls=[]
    def fake(method,ep,tok,data=None):
        calls.append((method,ep,tok,data))
        if ep=='/api/v1/me':
            return {'student':{'name':'Student 07'},'labs':[{'code':'KOOLAB_ELASTIC_SEC','state':{'milestones':[],'reported':0,'approved':0}}]}
        return {'ok':True}
    monkeypatch.setattr(tracker,'request_tracker',fake)
    h=login()
    r=client.post('/api/tracker/me',headers=h,json={'student_token':TOKEN})
    assert r.status_code==200,r.text
    assert r.json()['student']['name']=='Student 07'
    assert TOKEN not in r.text
    r=client.post('/api/tracker/start',headers=h,json={'student_token':TOKEN,'event_id':'1'*32})
    assert r.status_code==200,r.text
    assert calls[-1][1]=='/api/v1/events' and calls[-1][3]['event']=='started'
    assert calls[-1][2]==TOKEN

def test_https_still_supported(monkeypatch):
    monkeypatch.setattr(tracker,'request_tracker',lambda *a,**k: {'ok':True})
    h=login();h.update({'Origin':'https://testserver'})
    r=client.post('/api/tracker/heartbeat',headers=h,json={'student_token':TOKEN,'event_id':'2'*32})
    assert r.status_code==200,r.text

def test_reject_cross_origin_and_missing_origin(monkeypatch):
    monkeypatch.setattr(tracker,'request_tracker',lambda *a,**k: {'ok':True})
    h=login();h['Origin']='http://other-site.example'
    assert client.post('/api/tracker/start',headers=h,json={'student_token':TOKEN,'event_id':'3'*32}).status_code==403
    h.pop('Origin')
    assert client.post('/api/tracker/start',headers=h,json={'student_token':TOKEN,'event_id':'3'*32}).status_code==403

def test_health_probe_no_student_token(monkeypatch):
    monkeypatch.setattr(tracker,'probe_health',lambda:{'reachable':True,'upstream_status':200,'tracker':tracker.TRACKER_URL,'detail':'ok'})
    r=client.get('/api/tracker/diagnostics',headers=login())
    assert r.status_code==200,r.text
    assert r.json()['reachable'] is True
    assert TOKEN not in r.text
    assert client.get('/api/tracker/diagnostics').status_code==401

def test_cannot_claim_cloud_scaling_is_passed(monkeypatch):
    monkeypatch.setattr(tracker,'request_tracker',lambda *a,**k: {'ok':True})
    h=login()
    payload={'student_token':TOKEN,'milestone_key':'scaling_test','event':'passed','event_id':'4'*32}
    assert client.post('/api/tracker/event',headers=h,json=payload).status_code==422
    payload['event']='evidence'
    assert client.post('/api/tracker/event',headers=h,json=payload).status_code==200
    payload['event']='passed';payload['milestone_key']='rds_connected'
    assert client.post('/api/tracker/event',headers=h,json=payload).status_code==409

def test_tracker_url_still_requires_https(monkeypatch):
    monkeypatch.setattr(tracker,'TRACKER_URL','http://insecure-target.example')
    from fastapi import HTTPException
    try: tracker._base()
    except HTTPException as exc:assert exc.status_code==503
    else:assert False,'Expected HTTPS requirement for instructor relay'

def test_browser_js_http_uuid_fallback_exists():
    from pathlib import Path
    js=(Path(__file__).parent.parent/'app/static/app.js').read_text()
    assert 'newTrackerEventId' in js
    assert 'crypto.randomUUID()' not in js
    assert 'trackerConnect' in js
